### 题目 2：多节点系统资源均衡

bash

```bash
#!/bin/bash

# 节点列表
NODES=("node1" "node2" "node3" "node4" "node5")

# 函数：获取节点资源信息
get_node_resources() {
    local node=$1
    ssh $node "top -bn1 | grep 'Cpu(s)' | awk '{print \$2 + \$4}'; free | grep Mem | awk '{print \$3 * 100 / \$2}'; iostat -d 1 1 | tail -n +4 | awk '{print \$3 + \$4}'"
}

# 函数：迁移进程
migrate_process() {
    local from_node=$1
    local to_node=$2
    # 这里简单选择一个高 CPU 进程迁移，实际中需要更复杂的选择逻辑
    pid=$(ssh $from_node "ps -eo pid,%cpu --sort=-%cpu | head -n 2 | tail -n 1 | awk '{print \$1}'")
    ssh $from_node "gzip -c /proc/$pid/fd/0 | ssh $to_node 'gunzip -c > /tmp/migrated_process; chmod +x /tmp/migrated_process; nohup /tmp/migrated_process &'"
    ssh $from_node "kill -9 $pid"
}

while true; do
    declare -A cpu_usage
    declare -A mem_usage
    declare -A io_usage

    for node in "${NODES[@]}"; do
        read cpu mem io < <(get_node_resources $node)
        cpu_usage[$node]=$cpu
        mem_usage[$node]=$mem
        io_usage[$node]=$io
    done

    # 简单示例：根据 CPU 使用率迁移进程
    max_cpu_node=""
    max_cpu=0
    min_cpu_node=""
    min_cpu=100
    for node in "${NODES[@]}"; do
        if (( $(echo "${cpu_usage[$node]} > $max_cpu" | bc -l) )); then
            max_cpu=${cpu_usage[$node]}
            max_cpu_node=$node
        fi
        if (( $(echo "${cpu_usage[$node]} < $min_cpu" | bc -l) )); then
            min_cpu=${cpu_usage[$node]}
            min_cpu_node=$node
        fi
    done

    if (( $(echo "$max_cpu - $min_cpu > 20" | bc -l) )); then
        migrate_process $max_cpu_node $min_cpu_node
    fi

    sleep 60
done
```



此脚本需要在各节点配置好 SSH 免密登录，保存为 `resource_balance.sh`，添加执行权限后运行。

### 题目 3：内核参数动态调整

```python
#!/usr/bin/env python3
import psutil
import time
import subprocess

def adjust_kernel_params():
    # 监控网络流量
    net_io_counters = psutil.net_io_counters()
    bytes_sent = net_io_counters.bytes_sent
    bytes_recv = net_io_counters.bytes_recv
    time.sleep(1)
    net_io_counters = psutil.net_io_counters()
    new_bytes_sent = net_io_counters.bytes_sent
    new_bytes_recv = net_io_counters.bytes_recv
    sent_speed = new_bytes_sent - bytes_sent
    recv_speed = new_bytes_recv - bytes_recv
    total_speed = sent_speed + recv_speed

    if total_speed > 1024 * 1024 * 10:  # 流量超过 10MB/s
        subprocess.run(["sysctl", "net.core.somaxconn=4096"])
        subprocess.run(["sysctl", "net.ipv4.tcp_max_syn_backlog=4096"])
    else:
        subprocess.run(["sysctl", "net.core.somaxconn=128"])
        subprocess.run(["sysctl", "net.ipv4.tcp_max_syn_backlog=128"])

    # 监控内存使用率
    mem = psutil.virtual_memory()
    mem_percent = mem.percent
    if mem_percent > 80:
        subprocess.run(["sysctl", "vm.swappiness=20"])
    else:
        subprocess.run(["sysctl", "vm.swappiness=60"])

while True:
    adjust_kernel_params()
    time.sleep(5)
```

### 题目 8：多网络接口的故障切换

bash

```bash
#!/bin/bash

# 定义网络接口列表
INTERFACES=("eth0" "eth1" "eth2")

# 函数：检查接口状态
check_interface() {
    local interface=$1
    ping -c 1 -I $interface 8.8.8.8 > /dev/null 2>&1
    return $?
}

# 函数：切换网络流量
switch_interface() {
    local old_interface=$1
    local new_interface=$2
    ip route del default via $(ip route show dev $old_interface | grep default | awk '{print $3}')
    ip route add default via $(ip route show dev $new_interface | grep default | awk '{print $3}')
    echo "$(date): Switched from $old_interface to $new_interface" >> /var/log/network_switch.log
}

while true; do
    for interface in "${INTERFACES[@]}"; do
        if check_interface $interface; then
            current_interface=$interface
            break
        fi
    done

    for interface in "${INTERFACES[@]}"; do
        if [ "$interface" != "$current_interface" ]; then
            if check_interface $interface; then
                if ! check_interface $current_interface; then
                    switch_interface $current_interface $interface
                    current_interface=$interface
                fi
            fi
        fi
    done

    sleep 10
done
```

### 题目 10：恶意进程检测与隔离

```bash
#!/bin/bash

# 定义日志文件
LOG_FILE="/var/log/malware_detection.log"

# 函数：检测恶意进程
detect_malicious_process() {
    ps -eo pid,%cpu,cmd | while read pid cpu cmd; do
        if (( $(echo "$cpu > 80" | bc -l) )); then
            echo "$(date): High CPU usage process detected - PID: $pid, CPU: $cpu%, CMD: $cmd" >> $LOG_FILE
            isolate_process $pid
        fi
        netstat -anp | grep $pid | awk '{print $5}' | cut -d: -f1 | while read ip; do
            if grep -q "$ip" /etc/hosts.deny; then
                echo "$(date): Suspicious network connection process detected - PID: $pid, IP: $ip, CMD: $cmd" >> $LOG_FILE
                isolate_process $pid
            fi
        done
    done
}

# 函数：隔离进程
isolate_process() {
    local pid=$1
    kill -STOP $pid
    iptables -A INPUT -p all -s $(ps -p $pid -o comm=) -j DROP
    iptables -A OUTPUT -p all -d $(ps -p $pid -o comm=) -j DROP
}

while true; do
    detect_malicious_process
    sleep 60
done
```

### 题目一：批量文件重命名并分类归档

#### 题目描述



在一个指定目录下，有大量的图片文件（扩展名可能为 `.jpg`、`.png`、`.gif` 等），要求编写一个 Shell 脚本完成以下操作：



1. 按照图片的拍摄日期（从文件的元数据中获取）对图片进行分类。
2. 将同一日期拍摄的图片移动到以该日期命名的子目录中。
3. 重命名所有图片文件，新文件名格式为 `日期_序号.扩展名`，例如 `20250315_001.jpg`。

#### 代码实现



bash











```bash
#!/bin/bash

# 指定要处理的目录
target_dir="$1"

if [ -z "$target_dir" ]; then
    echo "Usage: $0 <directory>"
    exit 1
fi

# 检查目录是否存在
if [ ! -d "$target_dir" ]; then
    echo "Directory $target_dir does not exist."
    exit 1
fi

# 遍历目录下的所有图片文件
for file in "$target_dir"/*.{jpg,png,gif}; do
    if [ -f "$file" ]; then
        # 获取文件的拍摄日期
        date_taken=$(exiftool -CreateDate -s3 "$file" | tr -d ':' | cut -c1-8)
        if [ -z "$date_taken" ]; then
            date_taken="unknown"
        fi

        # 创建以日期命名的子目录
        date_dir="$target_dir/$date_taken"
        mkdir -p "$date_dir"

        # 统计该日期下已有文件数量
        count=$(ls "$date_dir" | wc -l)
        count=$((count + 1))
        new_count=$(printf "%03d" $count)

        # 获取文件扩展名
        ext="${file##*.}"

        # 生成新文件名
        new_name="${date_taken}_${new_count}.${ext}"

        # 移动并重命名文件
        mv "$file" "$date_dir/$new_name"
    fi
done
```

### 题目二：网络服务监控与自动重启

#### 题目描述



编写一个 Shell 脚本，用于监控指定的网络服务（如 Apache、Nginx 等）是否正常运行。如果服务停止响应，脚本应尝试自动重启服务，并记录重启操作到日志文件中。同时，若服务在短时间内多次重启失败，应发送邮件通知管理员。

#### 代码实现



bash











```bash
#!/bin/bash

# 要监控的服务名
service_name="nginx"
# 日志文件路径
log_file="/var/log/service_monitor.log"
# 邮件接收者
admin_email="admin@example.com"
# 最大重启次数
max_restarts=3
# 重启间隔（秒）
restart_interval=60
# 记录重启次数
restart_count=0

# 检查服务是否运行的函数
check_service() {
    systemctl is-active --quiet "$service_name"
    return $?
}

# 重启服务的函数
restart_service() {
    systemctl restart "$service_name"
    result=$?
    if [ $result -eq 0 ]; then
        echo "$(date): $service_name restarted successfully." >> "$log_file"
    else
        echo "$(date): Failed to restart $service_name." >> "$log_file"
    fi
    return $result
}

# 发送邮件通知的函数
```

4. 记录迁移过程中的详细信息，包括已迁移文件、迁移失败的文件等。

```bash
#!/bin/bash

# 源目录
SOURCE_DIR="/source/dir"
# 目标目录
TARGET_DIR="/target/dir"
# 日志文件
LOG_FILE="migration.log"
# 并行进程数
PROCESS_COUNT=4

# 筛选文件
filter_files() {
    find "$SOURCE_DIR" -type f -size +1M -mtime -7
}

# 复制文件
copy_file() {
    local file="$1"
    relative_path="${file#$SOURCE_DIR/}"
    target_file="$TARGET_DIR/$relative_path"
    target_dir=$(dirname "$target_file")
    mkdir -p "$target_dir"
    cp "$file" "$target_file"
    if [ $? -eq 0 ]; then
        echo "$(date): Successfully copied $file to $target_file" >> "$LOG_FILE"
    else
        echo "$(date): Failed to copy $file to $target_file" >> "$LOG_FILE"
    fi
}

# 多进程复制
parallel_copy() {
    local files=($(filter_files))
    local i=0
    while [ $i -lt ${#files[@]} ]; do
        for ((j = 0; j < $PROCESS_COUNT && i < ${#files[@]}; j++, i++)); do
            copy_file "${files[$i]}" &
        done
        wait
    done
}

# 验证文件完整性
verify_integrity() {
    local source_files=($(filter_files))
    for file in "${source_files[@]}"; do
        relative_path="${file#$SOURCE_DIR/}"
        target_file="$TARGET_DIR/$relative_path"
        if [ -f "$target_file" ]; then
            source_md5=$(md5sum "$file" | cut -d ' ' -f 1)

                echo "$(date): Integrity check failed for $file" >> "$LOG_FILE"
            fi
        else

        fi
    done
}
```

```bash
    local node="$1"
    if [ "$node" = "$MASTER_NODE" ]; then
        ssh -i "$SSH_KEY" "$node" "sudo systemctl start hadoop-master"
    else
        ssh -i "$SSH_KEY" "$node" "sudo systemctl start hadoop-slave"
    fi
}

# 检查服务状态
check_service_status() {
    local node="$1"
    ssh -i "$SSH_KEY" "$node" "sudo systemctl is-active hadoop"
}

# 部署集群
deploy_cluster() {
    for node in "${NODES[@]}"; do
        install_packages "$node"
        generate_config "$node"
        start_services "$node"
        status=$(check_service_status "$node")
        if [ "$status" = "active" ]; then
            echo "$node: Service started successfully"
        else
            echo "$node: Failed to start service"
        fi
    done
}
```
