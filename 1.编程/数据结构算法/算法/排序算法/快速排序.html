quickSort(A, lo, hi):
  if lo ≥ hi: return
  pivot = A[hi]          ← 选基准(橙)
  i = lo
  for j = lo … hi-1:     ← 比较(红)
    if A[j] < pivot:
      swap(A[i], A[j])
      i++
  swap(A[i], A[hi])      ← 基准归位(绿)
  quickSort(A, lo, i-1)  ← 递归左半
  quickSort(A, i+1, hi)  ← 递归右半
